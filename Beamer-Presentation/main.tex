\section{Innovation 2: BioHashing Cancelable Encryption}
% ============================================================================

\begin{frame}{BioHashing: Random Orthogonal Projection + Binarization}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{block}{Mathematical Foundation}
        \small
        Given raw biometric $X \in \mathbb{R}^n$, secret seed $K$, and random projection matrix $R \in \mathbb{R}^{m \times n}$ generated from $K$:
        \vspace{0.3em}

        \textbf{1. Projection:}
        \[ P_i = \sum_{j=1}^{n} R_{ij} \cdot X_j \quad \text{(dot product)} \]

        \textbf{2. Binarization:}
        \[ Y_i = \begin{cases} 1 & P_i \geq 0 \\ 0 & P_i < 0 \end{cases} \]

        \textbf{3. Matching (Hamming Distance):}
        \[ \text{Score} = 1 - \frac{d_H(Y_{\text{live}}, Y_{\text{stored}})}{512} \]
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{ISO/IEC 24745 Compliance}
        \small
        \begin{tabular}{lc}
          \toprule
          \textbf{Property} & \textbf{Status} \\
          \midrule
          Irreversibility & \textcolor{CodeGreen}{\faCheckCircle} \\
          Unlinkability & \textcolor{CodeGreen}{\faCheckCircle} \\
          Renewability & \textcolor{CodeGreen}{\faCheckCircle} \\
          \bottomrule
        \end{tabular}
      \end{block}
      \vspace{0.3em}
      \begin{alertblock}{Security Bounds}
        \small
        \textbf{Key space:} $2^{64} \approx 1.84 \times 10^{19}$\\
        \textbf{Template inversion:} $O(2^{512})$\\
        \textbf{Random collision:} $\approx 10^{-154}$
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
